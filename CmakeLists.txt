cmake_minimum_required(VERSION 3.14)
project(smart_pointers LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

option(BUILD_TESTS "Build func_tests" ON)
option(BUILD_BENCHMARKS "Build benchmarks" ON)

add_library(smart_pointers INTERFACE)
target_include_directories(smart_pointers INTERFACE ${CMAKE_SOURCE_DIR}/headers)

if(BUILD_TESTS)
    enable_testing()
    add_subdirectory(func_tests)
endif()

if(BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
